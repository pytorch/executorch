cmake_minimum_required(VERSION 3.24)
project(siliconbench_rewind_kv_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL TRUE)
set(EXECUTORCH_BUILD_CUDA ON CACHE BOOL "" FORCE)
set(MUSE_GLIMMER_BUILD_BATCHING ON CACHE BOOL "" FORCE)
set(MUSE_GLIMMER_BUILD_RUNTIME_KV_PROBE ON CACHE BOOL "" FORCE)
add_subdirectory("${ET_SOURCE}/examples/models/muse-glimmer" model)
get_target_property(_probe_links muse_glimmer_runtime_kv_probe LINK_LIBRARIES)
get_target_property(_probe_includes muse_glimmer_runtime_kv_probe INCLUDE_DIRECTORIES)
add_executable(rewind_kv_probe "${REWIND_PROBE_SOURCE}")
target_include_directories(rewind_kv_probe PRIVATE ${_probe_includes} "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(rewind_kv_probe PRIVATE ${_probe_links} CUDA::cudart)
target_link_options(rewind_kv_probe PRIVATE "LINKER:--export-dynamic")
