{
  global:
    JNI_OnLoad;
    Java_*;
    # A backend built as its own .so (EXECUTORCH_BUILD_XNNPACK_BACKEND_SHARED /
    # EXECUTORCH_BUILD_VULKAN_BACKEND_SHARED) carries its own statically linked
    # copy of executorch_core, and therefore its own backend registry. Its call
    # to register_backend goes through the PLT, so exporting the registry entry
    # points here lets the dynamic linker bind that call to this library's
    # registry - the one the runtime actually reads from.
    extern "C++" {
      executorch::runtime*::register_backend*;
      executorch::runtime*::get_backend_class*;
      executorch::runtime*::get_backend_name*;
      executorch::runtime*::get_num_registered_backends*;
    };
    # Same reason, for the thread pool the split XNNPACK backend runs on. It
    # reaches the pool through get_pthreadpool(), which owns the singleton, so
    # exporting only the C pthreadpool_* entry points would still leave the
    # backend with a second, unconfigured pool of its own; the accessors the
    # JNI layer configures through have to be exported too.
    extern "C++" {
      executorch::extension::threadpool*::get_pthreadpool*;
      executorch::extension::threadpool*::get_threadpool*;
    };
    pthreadpool_*;
  local:
    *;
};
